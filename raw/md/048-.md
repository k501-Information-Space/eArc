```
from __future__ import annotations

import json
import os
from pathlib import Path
from typing import Iterator
from pydantic import BaseModel

from k501.core.transition import Transition
from k501.core.validator import IdentityValidator
from k501.storage.exceptions import AppendFailureError, DuplicateTransitionError, TailCorruptionError

class JournalHead(BaseModel):
    last_valid_offset: int = 0
    record_count: int = 0
    last_transition_id: str | None = None

class JournalStore:
    def __init__(self, root: str | Path) -> None:
        self.root = Path(root)
        self.root.mkdir(parents=True, exist_ok=True)
        self.transitions_path = self.root / "transitions.jsonl"
        self.head_path = self.root / "head.json"
        
        if not self.transitions_path.exists():
            self.transitions_path.touch()

        # In-memory deduplication set
        self._seen_ids: set[str] = set()
        self._sync_state()

    def _sync_state(self) -> None:
        """Hydrates the known state directly from the authoritative journal log."""
        self._seen_ids.clear()
        valid_offset, count, last_id, _ = self._scan_journal()
        self._head = JournalHead(
            last_valid_offset=valid_offset,
            record_count=count,
            last_transition_id=last_id
        )
        self.persist_head()

    def load_head(self) -> JournalHead:
        if not self.head_path.exists():
            return JournalHead()
        with self.head_path.open("r", encoding="utf-8") as f:
            return JournalHead.model_validate_json(f.read())

    def persist_head(self) -> None:
        temp_path = self.head_path.with_suffix(".tmp")
        with temp_path.open("w", encoding="utf-8") as f:
            f.write(self._head.model_dump_json())
            f.flush()
            os.fsync(f.fileno())
        # Atomic rename
        temp_path.replace(self.head_path)

    def _scan_journal(self) -> tuple[int, int, str | None, bool]:
        """Returns (valid_offset, record_count, last_id, is_corrupted)."""
        valid_offset = 0
        count = 0
        last_id = None
        is_corrupted = False

        with self.transitions_path.open("rb") as f:
            while True:
                current_offset = f.tell()
                line = f.readline()
                if not line:
                    break
                
                try:
                    line_str = line.decode("utf-8").strip()
                    if not line_str:
                        continue
                        
                    data = json.loads(line_str)
                    t = Transition.model_validate(data)
                    
                    if not IdentityValidator.validate(t).valid:
                        is_corrupted = True
                        break

                    self._seen_ids.add(t.transition_id)
                    valid_offset = f.tell()
                    count += 1
                    last_id = t.transition_id

                except (json.JSONDecodeError, UnicodeDecodeError, ValueError):
                    is_corrupted = True
                    break

        return valid_offset, count, last_id, is_corrupted

    def detect_tail_corruption(self) -> bool:
        _, _, _, is_corrupted = self._scan_journal()
        if is_corrupted:
            return True
        
        # Check for uncommitted dangling bytes
        actual_size = self.transitions_path.stat().st_size
        return actual_size > self._head.last_valid_offset

    def recover_tail(self) -> None:
        if not self.detect_tail_corruption():
            return
            
        valid_offset, count, last_id, _ = self._scan_journal()
        
        # Hard truncate dropping the corrupted trailing bytes
        with self.transitions_path.open("r+b") as f:
            f.truncate(valid_offset)
            f.flush()
            os.fsync(f.fileno())
            
        self._sync_state()

    def append_transition(self, transition: Transition) -> str:
        if transition.transition_id in self._seen_ids:
            raise DuplicateTransitionError(f"Duplicate ID: {transition.transition_id}")

        val_result = IdentityValidator.validate(transition)
        if not val_result.valid:
            raise AppendFailureError("Transition identity validation failed.")

        payload = transition.model_dump_json().encode("utf-8") + b"\n"

        try:
            # Append in binary mode to ensure exact byte writing
            with self.transitions_path.open("ab") as f:
                f.write(payload)
                f.flush()
                os.fsync(f.fileno())  # Strict durability barrier
                
                # Update State
                self._head.last_valid_offset += len(payload)
                self._head.record_count += 1
                self._head.last_transition_id = transition.transition_id
                
        except Exception as e:
            # Re-raise strict append failure, don't swallow
            raise AppendFailureError(f"Failed to append to journal: {str(e)}") from e

        self._seen_ids.add(transition.transition_id)
        self.persist_head()
        return transition.transition_id

    def read_transitions(self) -> Iterator[Transition]:
        with self.transitions_path.open("rb") as f:
            for line in f:
                line_str = line.decode("utf-8").strip()
                if line_str:
                    yield Transition.model_validate_json(line_str)
```

```
def test_journal_append_and_read(tmp_path):

    t1 = create_dummy_transition(1000, "active")

    store.append_transition(t1)

    assert len(records) == 1

    store = JournalStore(root=tmp_path / "journal")
    t1 = create_dummy_transition(1000, "active")
    t2 = create_dummy_transition(2000, "archived")
    
    store.append_transition(t1)
    store.append_transition(t2)
    
    assert store.load_head().record_count == 2
    
    # 3. Simulate partial write / tail corruption
    with store.transitions_path.open("ab") as f:

    # 5. Recovery drops the bad bytes and preserves the valid ones
    store.recover_tail()
    assert store.detect_tail_corruption() is False
    assert store.load_head().record_count == 2
    
    records = list(store.read_transitions())
    assert len(records) == 2
    assert records[-1].transition_id == t2.transition_id
```

```
import os
from pathlib import Path
from typing import Iterator
from pydantic import BaseModel

from k501.core.transition import Transition
from k501.core.validator import IdentityValidator
from k501.storage.exceptions import AppendFailureError, DuplicateTransitionError

class JournalHead(BaseModel):
    last_valid_offset: int = 0
    record_count: int = 0
    last_transition_id: str | None = None

class JournalStore:

        self.root = Path(root)
        self.root.mkdir(parents=True, exist_ok=True)
        self.transitions_path = self.root / "transitions.jsonl"
        self.head_path = self.root / "head.json"
        
        if not self.transitions_path.exists():
            self.transitions_path.touch()

        self._seen_ids: set[str] = set()
        self._sync_state()

    def _sync_state(self) -> None:
        """Hydrates the known state by parsing the authoritative log."""
        self._seen_ids.clear()
        valid_offset, count, last_id, _ = self._scan_journal()
        self._head = JournalHead(
            last_valid_offset=valid_offset,
            record_count=count,
            last_transition_id=last_id
        )
        self.persist_head()

    def load_head(self) -> JournalHead:
        if not self.head_path.exists():
            return JournalHead()
        with self.head_path.open("r", encoding="utf-8") as f:
            return JournalHead.model_validate_json(f.read())

    def persist_head(self) -> None:
        temp_path = self.head_path.with_suffix(".tmp")
        with temp_path.open("w", encoding="utf-8") as f:
            f.write(self._head.model_dump_json())
            f.flush()
            os.fsync(f.fileno())
        temp_path.replace(self.head_path)

    def _scan_journal(self) -> tuple[int, int, str | None, bool]:
        """Returns (valid_offset, record_count, last_id, is_corrupted)."""
        valid_offset = 0
        count = 0
        last_id = None
        is_corrupted = False

        with self.transitions_path.open("rb") as f:
            while True:
                line = f.readline()
                if not line:
                    break
                
                try:
                    line_str = line.decode("utf-8").strip()
                    if not line_str:
                        continue
                        
                    data = json.loads(line_str)
                    t = Transition.model_validate(data)
                    
                    if not IdentityValidator.validate(t).valid:
                        is_corrupted = True
                        break

                    self._seen_ids.add(t.transition_id)
                    valid_offset = f.tell()
                    count += 1
                    last_id = t.transition_id

                except (json.JSONDecodeError, UnicodeDecodeError, ValueError):
                    is_corrupted = True
                    break

        return valid_offset, count, last_id, is_corrupted

    def detect_tail_corruption(self) -> bool:
        _, _, _, is_corrupted = self._scan_journal()
        if is_corrupted:
            return True
        
        actual_size = self.transitions_path.stat().st_size
        return actual_size > self._head.last_valid_offset

    def recover_tail(self) -> None:
        if not self.detect_tail_corruption():
            return
            
        valid_offset, count, last_id, _ = self._scan_journal()
        
        # Hard truncate dropping the corrupted trailing bytes
        with self.transitions_path.open("r+b") as f:
            f.truncate(valid_offset)
            f.flush()
            os.fsync(f.fileno())
            
        self._sync_state()

    def append_transition(self, transition: Transition) -> str:
        if transition.transition_id in self._seen_ids:
            raise DuplicateTransitionError(f"Duplicate ID: {transition.transition_id}")

        val_result = IdentityValidator.validate(transition)
        if not val_result.valid:
            raise AppendFailureError("Transition identity validation failed.")

        # Write as binary to strictly align byte offsets for the indexer
        payload = transition.model_dump_json().encode("utf-8") + b"\n"

        try:
            with self.transitions_path.open("ab") as f:
                f.write(payload)
                f.flush()
                os.fsync(f.fileno())
                
                self._head.last_valid_offset += len(payload)
                self._head.record_count += 1
                self._head.last_transition_id = transition.transition_id
                
        except Exception as e:
            raise AppendFailureError(f"Failed to append to journal: {str(e)}") from e

        self._seen_ids.add(transition.transition_id)
        self.persist_head()
        return transition.transition_id

    def read_transitions(self) -> Iterator[Transition]:
        with self.transitions_path.open("rb") as f:
            for line in f:
                line_str = line.decode("utf-8").strip()
                if line_str:
                    yield Transition.model_validate_json(line_str)
```

```
                line = f.readline()
                if not line:
                    break
                    
                line_str = line.decode("utf-8").strip()
                if not line_str:
                    continue
                    
                try:
                    data = json.loads(line_str)
                    t_id = data.get("transition_id")
                    if t_id:
                        self.entries[t_id] = IndexEntry(
                            transition_id=t_id,
                            offset=start_offset,
                            length=len(line)
                        )
                except json.JSONDecodeError:
                    break
```
