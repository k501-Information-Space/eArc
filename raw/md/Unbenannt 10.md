`""" Graph Store: ProvenanceGraph implementation. Provides persistence and traversal for the provenance graph. """ import json import sqlite3 from datetime import datetime from pathlib import Path from typing import Any, Dict, List, Optional from .models import Node, Edge, NodeType, EdgeType class ProvenanceGraph:     """    Persistent provenance graph with SQLite backend.         Supports:    - Node and edge persistence    - Lineage traversal    - Provenance chain reconstruction    - Temporal and namespace filtering    """         def __init__(self, db_path: str = "data/graph/graph.sqlite"):        """Initialize the graph with SQLite backend."""        self.db_path = Path(db_path)        self.db_path.parent.mkdir(parents=True, exist_ok=True)        self._init_db()         def _init_db(self) -> None:        """Initialize SQLite database schema."""        conn = sqlite3.connect(str(self.db_path))        cursor = conn.cursor()                 # Nodes table        cursor.execute("""            CREATE TABLE IF NOT EXISTS nodes (                node_id TEXT PRIMARY KEY,                node_type TEXT NOT NULL,                state_ref TEXT NOT NULL,                attributes TEXT NOT NULL,                created_at TEXT NOT NULL            )        """)                 # Edges table        cursor.execute("""            CREATE TABLE IF NOT EXISTS edges (                source_id TEXT NOT NULL,                edge_type TEXT NOT NULL,                target_id TEXT NOT NULL,                attributes TEXT NOT NULL,                created_at TEXT NOT NULL,                PRIMARY KEY (source_id, edge_type, target_id)            )        """)                 # Indexes for performance        cursor.execute("""            CREATE INDEX IF NOT EXISTS idx_edges_source            ON edges(source_id)        """)        cursor.execute("""            CREATE INDEX IF NOT EXISTS idx_edges_target            ON edges(target_id)        """)        cursor.execute("""            CREATE INDEX IF NOT EXISTS idx_edges_type            ON edges(edge_type)        """)                 conn.commit()        conn.close()         def add_node(self, node: Node) -> bool:        """Add a node to the graph."""        try:            conn = sqlite3.connect(str(self.db_path))            cursor = conn.cursor()                         cursor.execute("""                INSERT OR REPLACE INTO nodes                (node_id, node_type, state_ref, attributes, created_at)                VALUES (?, ?, ?, ?, ?)            """, (                node.node_id,                node.node_type.value,                node.state_ref,                json.dumps(node.attributes),                node.created_at.isoformat(),            ))                         conn.commit()            conn.close()            return True        except Exception as e:            print(f"Error adding node: {e}")            return False         def add_edge(self, edge: Edge) -> bool:        """Add an edge to the graph."""        try:            conn = sqlite3.connect(str(self.db_path))            cursor = conn.cursor()                         cursor.execute("""                INSERT OR REPLACE INTO edges                (source_id, edge_type, target_id, attributes, created_at)                VALUES (?, ?, ?, ?, ?)            """, (                edge.source_id,                edge.edge_type.value,                edge.target_id,                json.dumps(edge.attributes),                edge.created_at.isoformat(),            ))                         conn.commit()            conn.close()            return True        except Exception as e:            print(f"Error adding edge: {e}")            return False         def get_node(self, node_id: str) -> Optional[Node]:        """Get a node by ID."""        conn = sqlite3.connect(str(self.db_path))        cursor = conn.cursor()                 cursor.execute("""            SELECT node_id, node_type, state_ref, attributes, created_at            FROM nodes WHERE node_id = ?        """, (node_id,))                 row = cursor.fetchone()        conn.close()                 if row is None:            return None                 return Node(            node_id=row[0],            node_type=NodeType(row[1]),            state_ref=row[2],            attributes=json.loads(row[3]),            created_at=datetime.fromisoformat(row[4]),        )         def get_edges(        self,        node_id: str,        direction: str = "forward",        edge_types: Optional[List[EdgeType]] = None,    ) -> List[Edge]:        """Get all edges for a node."""        conn = sqlite3.connect(str(self.db_path))        cursor = conn.cursor()                 if direction == "forward":            cursor.execute("""                SELECT source_id, edge_type, target_id, attributes, created_at                FROM edges WHERE source_id = ?            """, (node_id,))        else:  # backward            cursor.execute("""                SELECT source_id, edge_type, target_id, attributes, created_at                FROM edges WHERE target_id = ?            """, (node_id,))                 rows = cursor.fetchall()        conn.close()                 edges = []        for row in rows:            if edge_types is None or EdgeType(row[1]) in edge_types:                edges.append(Edge(                    source_id=row[0],                    edge_type=EdgeType(row[1]),                    target_id=row[2],                    attributes=json.loads(row[3]),                    created_at=datetime.fromisoformat(row[4]),                ))                 return edges         def traverse(        self,        start: str,        edge_types: List[EdgeType],        direction: str = "backward",        depth: int = 10,        time_filter: Optional[str] = None,        namespace_filter: Optional[str] = None,    ) -> List[str]:        """        Traverse the graph with filters.                 Args:            start: Starting node ID            edge_types: List of edge types to follow            direction: "forward" or "backward"            depth: Maximum traversal depth            time_filter: ISO8601 timestamp to filter by            namespace_filter: Namespace to filter by                 Returns:            List of node IDs in traversal order        """        visited = set()        result = []        current_level = [start]                 for _ in range(depth):            next_level = []                         for node_id in current_level:                if node_id in visited:                    continue                                 visited.add(node_id)                result.append(node_id)                                 # Get edges                edges = self.get_edges(node_id, direction, edge_types)                                 for edge in edges:                    # Apply filters                    if time_filter:                        edge_time = edge.attributes.get("created_at")                        if edge_time and edge_time > time_filter:                            continue                                         if namespace_filter:                        node = self.get_node(edge.target_id if direction == "forward" else edge.source_id)                        if node and node.attributes.get("namespace") != namespace_filter:                            continue                                         # Add to next level                    target = edge.target_id if direction == "forward" else edge.source_id                    if target not in visited:                        next_level.append(target)                         current_level = next_level            if not current_level:                break                 return result         def lineage(self, object_id: str) -> List[str]:        """        Get lineage chain for an object.                 Returns transitive ancestry chain.        """        return self.traverse(            start=object_id,            edge_types=[EdgeType.DERIVED_FROM, EdgeType.ANCESTOR],            direction="backward",            depth=100,        )         def provenance(self, object_id: str) -> Dict[str, List[str]]:        """        Get proof and audit chain for an object.                 Returns dict with proof_ids and audit_ids.
