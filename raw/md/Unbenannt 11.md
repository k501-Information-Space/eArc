`""" Canonical Serialization: Deterministic JSON serialization. Ensures that identical state objects produce identical byte sequences, which is required for deterministic ID generation. """ import json from typing import Any, Dict def canonical_serialize(obj: Any) -> str:     """    Serialize an object to canonical JSON.         Rules:    - Keys are sorted alphabetically    - No trailing whitespace    - No indentation    - Unicode preserved    - Floats normalized to 6 decimal places    """    if isinstance(obj, dict):        # Sort keys recursively        sorted_dict = {k: canonical_serialize(v) for k, v in sorted(obj.items())}        return json.dumps(sorted_dict, sort_keys=True, separators=(',', ':'), ensure_ascii=False)    elif isinstance(obj, list):        return json.dumps([canonical_serialize(item) for item in obj], separators=(',', ':'), ensure_ascii=False)    elif isinstance(obj, float):        # Normalize floats to 6 decimal places        return f"{obj:.6f}"    elif isinstance(obj, bool):        return json.dumps(obj)    elif isinstance(obj, None):        return json.dumps(obj)    else:        # String, int, etc.        return json.dumps(obj, ensure_ascii=False) def canonical_hash(obj: Any) -> str:     """    Create a canonical hash of an object.         First serializes canonically, then hashes with SHA-256.    """    import hashlib    canonical_str = canonical_serialize(obj)    hash_bytes = hashlib.sha256(canonical_str.encode()).digest()    return hash_bytes.hex()`
