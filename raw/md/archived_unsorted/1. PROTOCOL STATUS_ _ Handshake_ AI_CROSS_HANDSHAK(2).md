#!/usr/bin/env python3  
"""  
K501 Gaia Loop Module (Transition Operator T)  
Role: Iterates the weighted dynamical graph from S_t to S_{t+1}.  
Hardware Constraint: 4 Threads / 8GB RAM (ThinkPad T520) - Enforced via sleep().  
"""  
import json  
import os  
import sys  
import urllib.request  
import time  
  
# I/O Paths  
PATH_RAW_FRAMES = "/home/iinkognit0/k501_llm_test/in/batch_002.ndjson"  
PATH_STATE_T0 = "/home/iinkognit0/workspace/archive/little_alpha_storage/master_index_v1.ndjson"  
PATH_STATE_T1 = "/home/iinkognit0/workspace/archive/little_alpha_storage/master_index_v2_t1.ndjson"  
  
# API Configuration  
OLLAMA_URL = "http://localhost:11434/api/generate"  
MODEL_ID = "qwen2:0.5b-instruct"  
  
def load_anchor_content(filepath: str) -> dict:  
"""Loads raw frame content to serve as the invariant anchor."""  
anchor_map = {}  
if not os.path.exists(filepath):  
print(f"[FATAL] Anchor file not found: {filepath}", file=sys.stderr)  
sys.exit(1)  
with open(filepath, 'r') as f:  
for line in f:  
try:  
data = json.loads(line)  
anchor_map[data.get("id")] = data.get("content", "")  
except json.JSONDecodeError:  
pass  
return anchor_map  
  
def calculate_mass(labels_t0: list, labels_t1: list) -> float:  
"""Calculates semantic density m_i(t) based on label intersection/union."""  
set_t0 = set(labels_t0)  
set_t1 = set(labels_t1)  
union_len = len(set_t0.union(set_t1))  
if union_len == 0:  
return 1.0 # Absolute stability (empty sets)  
diff_len = len(set_t1.symmetric_difference(set_t0))  
return 1.0 - (diff_len / union_len)  
  
def execute_t_operator():  
print("[SYS] Initializing Transition Operator T...")  
anchor_map = load_anchor_content(PATH_RAW_FRAMES)  
  
# Initialize Bootframe System Instruction  
sys_instruction = (  
"Evaluate S_t against raw frame content. Purge contradictions. Condense invariants. "  
"Output STRICT JSON format: {\"labels\": [\"tag1\", \"tag2\"], \"type\": \"doc_type\"}. "  
"LOWERCASE ONLY. NO EXPLANATIONS."  
)  
  
processed_nodes = 0  
  
with open(PATH_STATE_T0, 'r') as f_in, open(PATH_STATE_T1, 'w') as f_out:  
for line in f_in:  
try:  
node_t0 = json.loads(line)  
except json.JSONDecodeError:  
continue  
  
node_id = node_t0.get("id", "unknown")  
labels_t0 = node_t0.get("semantic", {}).get("labels", [])  
anchor_content = anchor_map.get(node_id, "")  
  
# Base Case definition (t=0 implies no prior mass tracking)  
current_mass = 0.0  
  
# Construct Bootframe Prompt  
prompt = (  
f"{sys_instruction} "  
f"ANCHOR_CONTENT: {anchor_content[:400]} " # Bounded to prevent context overflow  
f"PRIOR_LABELS_T0: {labels_t0} "  
f"CURRENT_MASS: {current_mass}"  
)  
  
# Query Local API  
payload = json.dumps({  
"model": MODEL_ID,  
"prompt": prompt,  
"stream": False  
}).encode('utf-8')  
  
req = urllib.request.Request(OLLAMA_URL, data=payload, headers={'Content-Type': 'application/json'})  
  
try:  
with urllib.request.urlopen(req) as response:  
api_resp = json.loads(response.read().decode('utf-8'))  
llm_output_str = api_resp.get("response", "{}")  
  
# Attempt to extract JSON payload from LLM response  
try:  
llm_data_t1 = json.loads(llm_output_str)  
except json.JSONDecodeError:  
llm_data_t1 = {"labels": labels_t0, "type": node_t0.get("semantic", {}).get("type", "unknown")}  
print(f"[WARN] Fallback invoked for node: {node_id}", file=sys.stderr)  
  
except Exception as e:  
print(f"[ERR] API Failure on node {node_id}: {str(e)}", file=sys.stderr)  
continue  
  
# State Transition Math  
labels_t1 = llm_data_t1.get("labels", [])  
if not isinstance(labels_t1, list):  
labels_t1 = []  
  
new_mass = calculate_mass(labels_t0, labels_t1)  
  
# Construct S_{t+1} Node  
node_t1 = {  
"id": node_id,  
"t": node_t0.get("t", 0),  
"h": node_t0.get("h", "unknown"),  
"semantic": {  
"labels": labels_t1,  
"type": llm_data_t1.get("type", "unknown")  
},  
"mass": round(new_mass, 4),  
"vector": node_t0.get("vector", [])  
}  
  
f_out.write(json.dumps(node_t1) + "\n")  
processed_nodes += 1  
print(f"[GAIA] Iterated {node_id} | Mass: {current_mass} -> {round(new_mass, 4)}")  
  
# Hardware Protection Pause (ThinkPad T520)  
time.sleep(0.5)  
  
print(f"[SYS] Transition S_0 -> S_1 Complete. Nodes processed: {processed_nodes}")  
  
if __name__ == "__main__":  
execute_t_operator()
