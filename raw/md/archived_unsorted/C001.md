/* ============================================================

K501 · Ultra-Optimized Kernel V2

Target: 300–500 Frames/sec on ThinkPad T520 (Sandy Bridge)

Components implemented in this reference code:

- frame_engine_fast

- hash_pipeline (OpenSSL EVP SHA256)

- ledger_writer (buffered append)

- minimal frame structure

Compile:

gcc -O3 -march=sandybridge -mtune=sandybridge -fomit-frame-pointer \

-flto -Wall -Wextra -I./include \

k501_kernel_v2.c -lcrypto -o k501_kernel_v2

============================================================ */

#include <stdio.h>

#include <stdlib.h>

#include <stdint.h>

#include <string.h>

#include <unistd.h>

#include <fcntl.h>

#include <time.h>

#include <openssl/evp.h>

/* ============================================================

CONFIG

============================================================ */

#define FRAME_BUFFER_SIZE 262144

#define MAX_FRAME_SIZE 1024

#define LEDGER_FILE "frames.ndjson"

/* ============================================================

GLOBAL LEDGER WRITER

============================================================ */

static int ledger_fd = -1;

static char write_buffer[FRAME_BUFFER_SIZE];

static size_t write_pos = 0;

/* ============================================================

HASH PIPELINE

============================================================ */

void sha256_hash(const unsigned char *data, size_t len, unsigned char *out_hash)

{

EVP_MD_CTX *ctx = EVP_MD_CTX_new();

EVP_DigestInit_ex(ctx, EVP_sha256(), NULL);

EVP_DigestUpdate(ctx, data, len);

EVP_DigestFinal_ex(ctx, out_hash, NULL);

EVP_MD_CTX_free(ctx);

}

/* ============================================================

LEDGER WRITER

============================================================ */

void ledger_flush()

{

if(write_pos > 0)

{

write(ledger_fd, write_buffer, write_pos);

write_pos = 0;

}

}

void ledger_append(const char *frame, size_t len)

{

if(write_pos + len >= FRAME_BUFFER_SIZE)

{

ledger_flush();

}

memcpy(write_buffer + write_pos, frame, len);

write_pos += len;

}

/* ============================================================

FRAME ENGINE FAST

============================================================ */

void build_frame(char *out, const char *id, const char *hash_hex, uint64_t timestamp)

{

/* fixed template avoids heavy sprintf logic */

char *ptr = out;

ptr += sprintf(ptr, "{\"id\":\"");

ptr += sprintf(ptr, "%s", id);

ptr += sprintf(ptr, "\",\"t\":");

ptr += sprintf(ptr, "%lu", timestamp);

ptr += sprintf(ptr, ",\"hash\":\"");

ptr += sprintf(ptr, "%s", hash_hex);

ptr += sprintf(ptr, "\"}\n");

}

/* ============================================================

HASH → HEX

============================================================ */

void hash_to_hex(const unsigned char *hash, char *hex)

{

static const char *table = "0123456789abcdef";

for(int i = 0; i < 32; i++)

{

hex[i*2] = table[(hash[i] >> 4) & 0xF];

hex[i*2+1] = table[hash[i] & 0xF];

}

hex[64] = 0;

}

/* ============================================================

FRAME EMIT

============================================================ */

void emit_frame(uint64_t seq)

{

char frame[MAX_FRAME_SIZE];

char id[64];

char hash_hex[65];

unsigned char hash[32];

snprintf(id, sizeof(id), "FRAME-%lu", seq);

sha256_hash((unsigned char*)id, strlen(id), hash);

hash_to_hex(hash, hash_hex);

uint64_t now = time(NULL);

build_frame(frame, id, hash_hex, now);

ledger_append(frame, strlen(frame));

}

/* ============================================================

MAIN KERNEL LOOP

============================================================ */

int main()

{

ledger_fd = open(LEDGER_FILE, O_CREAT | O_APPEND | O_WRONLY, 0644);

if(ledger_fd < 0)

{

perror("ledger open");

return 1;

}

printf("K501 Kernel V2 starting...\n");

uint64_t seq = 0;

while(seq < 1000000)

{

emit_frame(seq);

seq++;

/* optional throttle for stability testing */

/* usleep(1000); */

}

ledger_flush();

close(ledger_fd);

printf("Completed.\n");

return 0;

}
