## `index.sh` — Minimaler, stabiler Indexer

```sh
#!/bin/sh
# ============================================================
# GX / eArc · Minimal Index Generator
# MODE: READ-ONLY · NO MUTATION
# ============================================================

ROOT_DIR="${1:-.}"
OUT_FILE="index.ndjson"
TS_UTC="$(date -u +"%Y-%m-%dT%H:%M:%SZ")"
TS_UNIX="$(date -u +"%s")"

echo "# INDEX GENERATED" > "$OUT_FILE"
echo "# UTC  $TS_UTC" >> "$OUT_FILE"
echo "# UNIX $TS_UNIX" >> "$OUT_FILE"

find "$ROOT_DIR" -type f ! -path "./$OUT_FILE" | while read -r FILE; do
  SIZE="$(stat -c %s "$FILE" 2>/dev/null || stat -f %z "$FILE")"
  HASH="$(sha256sum "$FILE" | awk '{print $1}')"

  printf '{' >> "$OUT_FILE"
  printf '"path":"%s",' "$(printf "%s" "$FILE" | sed 's/"/\\"/g')" >> "$OUT_FILE"
  printf '"bytes":%s,' "$SIZE" >> "$OUT_FILE"
  printf '"sha256":"%s",' "$HASH" >> "$OUT_FILE"
  printf '"ts_utc":"%s",' "$TS_UTC" >> "$OUT_FILE"
  printf '"ts_unix":%s' "$TS_UNIX" >> "$OUT_FILE"
  printf '}\n' >> "$OUT_FILE"
done

echo "# DONE" >> "$OUT_FILE"
```
