def _fetchall(con: sqlite3.Connection, sql: str, params: Tuple[Any, ...] = ()) -> List[Dict[str, Any]]:
    cur = con.execute(sql, params)
    rows = cur.fetchall()
    return [dict(r) for r in rows]

@app.get("/packs/{pack_id}")
def pack_by_id(pack_id: str, include_items: bool = Query(default=True), db_path: str = Query(default=DEFAULT_DB_PATH)) -> Dict[str, Any]:
    con = _connect(db_path)
    try:
        pack = _fetchone(con, """
            SELECT id, title, state, created_at, fixed_at

            WHERE id = ?
        """, (pack_id,))

        return pack
    finally:
        con.close()
