<?php
/**
 * Plugin Name: eArc RSS (Canonical Aggregator)
 * Description: Provides canonical /rss feed that aggregates Frames + Substack + Podcast RSS.
 */

if (!defined('ABSPATH')) exit;

const EARC_CANONICAL_RSS_PATH = '/rss';
const EARC_CACHE_KEY = 'earc_rss_cache_v1';
const EARC_CACHE_TTL = 15 * MINUTE_IN_SECONDS;

// Your source feeds (canonical)
const EARC_FEED_SUBSTACK = 'https://iinkognit0.substack.com/feed';
const EARC_FEED_PODCAST  = 'https://anchor.fm/s/10d156d24/podcast/rss';

// Optional: local Frames posts category slug
const EARC_FRAMES_CATEGORY = 'frames';

add_action('init', function () {
  // Provide feed endpoint: /?feed=earc and pretty URL /rss
  add_feed('earc', 'earc_render_feed');
  add_rewrite_rule('^rss/?$', 'index.php?feed=earc', 'top');
});

// Make /feed/ redirect to /rss for canonicality (optional but recommended)
add_action('template_redirect', function () {
  if (is_feed() && !is_feed('earc')) {
    // Redirect only the main site RSS feed; leave other feeds alone
    if (get_query_var('feed') === 'rss2' || get_query_var('feed') === 'feed') {
      wp_redirect(home_url(EARC_CANONICAL_RSS_PATH), 301);
      exit;
    }
  }
});

// Render the canonical eArc feed
function earc_render_feed() {
  header('Content-Type: application/rss+xml; charset=UTF-8');

  $xml = get_transient(EARC_CACHE_KEY);
  if ($xml) {
    echo $xml;
    return;
  }

  $items = [];

  // 1) Frames (local WP posts in category "frames") - optional but supported
  $frames_posts = get_posts([
    'post_type'      => 'post',
    'posts_per_page' => 20,
    'post_status'    => 'publish',
    'tax_query'      => [[
      'taxonomy' => 'category',
      'field'    => 'slug',
      'terms'    => [EARC_FRAMES_CATEGORY],
    ]],
  ]);

  foreach ($frames_posts as $p) {
    $items[] = [
      'title' => get_the_title($p),
      'link'  => get_permalink($p),
      'date'  => get_post_time('U', true, $p),
      'desc'  => wp_strip_all_tags(get_the_excerpt($p)),
      'src'   => 'frames',
    ];
  }

  // 2) External feeds (Substack + Podcast master feed)
  $ext = earc_fetch_external_items([
    ['url' => EARC_FEED_SUBSTACK, 'src' => 'substack'],
    ['url' => EARC_FEED_PODCAST,  'src' => 'podcast'],
  ]);

  $items = array_merge($items, $ext);

  // Sort newest first
  usort($items, fn($a, $b) => ($b['date'] <=> $a['date']));

  // Limit total items
  $items = array_slice($items, 0, 30);

  // Build RSS XML (self link = /rss)
  $site_title = 'iInkognit0';
  $site_link  = home_url('/');
  $site_desc  = 'Das Ewige Archiv';
  $self_link  = home_url(EARC_CANONICAL_RSS_PATH);
  $build_date = gmdate('D, d M Y H:i:s +0000');

  $out  = '<?xml version="1.0" encoding="UTF-8"?>' . "\n";
  $out .= '<rss version="2.0" xmlns:atom="http://www.w3.org/2005/Atom">' . "\n";
  $out .= "<channel>\n";
  $out .= '  <title>' . earc_xml($site_title) . "</title>\n";
  $out .= '  <atom:link href="' . earc_xml($self_link) . '" rel="self" type="application/rss+xml" />' . "\n";
  $out .= '  <link>' . earc_xml($site_link) . "</link>\n";
  $out .= '  <description>' . earc_xml($site_desc) . "</description>\n";
  $out .= '  <language>de</language>' . "\n";
  $out .= '  <lastBuildDate>' . $build_date . "</lastBuildDate>\n";

  foreach ($items as $it) {
    $out .= "  <item>\n";
    $out .= '    <title>' . earc_xml($it['title']) . "</title>\n";
    $out .= '    <link>' . earc_xml($it['link']) . "</link>\n";
    $out .= '    <pubDate>' . gmdate('D, d M Y H:i:s +0000', (int)$it['date']) . "</pubDate>\n";
    $out .= '    <description><![CDATA[' . earc_cdata($it['desc']) . "]]></description>\n";
    $out .= '    <guid isPermaLink="false">' . earc_xml(md5($it['link'] . '|' . $it['date'])) . "</guid>\n";
    $out .= "  </item>\n";
  }

  $out .= "</channel>\n</rss>\n";

  set_transient(EARC_CACHE_KEY, $out, EARC_CACHE_TTL);
  echo $out;
}

function earc_fetch_external_items(array $feeds) : array {
  if (!function_exists('fetch_feed')) {
    require_once ABSPATH . WPINC . '/feed.php';
  }

  $items = [];

  foreach ($feeds as $f) {
    $feed = fetch_feed($f['url']);
    if (is_wp_error($feed)) continue;

    $max = min(20, (int)$feed->get_item_quantity(20));
    foreach ($feed->get_items(0, $max) as $it) {
      $title = trim((string)$it->get_title());
      $link  = trim((string)$it->get_link());
      $date  = (int)$it->get_date('U');
      $desc  = trim(wp_strip_all_tags((string)$it->get_description()));

      if (!$title || !$link || !$date) continue;

      $items[] = [
        'title' => $title,
        'link'  => $link,
        'date'  => $date,
        'desc'  => $desc,
        'src'   => $f['src'],
      ];
    }
  }

  return $items;
}

function earc_xml(string $s) : string {
  return htmlspecialchars($s, ENT_XML1 | ENT_COMPAT, 'UTF-8');
}

function earc_cdata(string $s) : string {
  // Prevent closing CDATA injection
  return str_replace(']]>', ']]]]><![CDATA[>', $s);
}
