```js
        receipt.ts_unix,
        receipt.id
      ]);
    }

    await conn.commit();
  }catch(e){
    await conn.rollback();
    throw e;
  }finally{
    conn.release();
  }
}
```

## 3) Handler-Patch (Minimum Hardening)

```js
  }
```
