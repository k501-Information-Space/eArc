data class Frame(val id: String, val utc: String, val source: String, val type: String, val artifact: ArtifactRef?)

fun ingest(stream: InputStream, mime: String, source: String) {
  val utc = Instant.now().toString()

  val artifact = storage.storeArtifact(stream, mime, utc) // returns ArtifactRef
  val frame = frameFactory.newFrame(utc, source, "artifact_ingest", artifact)

  val framePath = storage.writeFrame(frame)

  storage.appendLog("analysis.log.ndjson",
    """{"utc":"$utc","event":"FRAME_DEFINED","frame_id":"${frame.id}"}"""
  )
  storage.appendLog("io.log.ndjson",
