#!/usr/bin/env python3
"""
Essay → FrameCode (IAL) Compiler v0.1.1 (MINIMAL, REFINED)

Refinements:
1) Saubere Trennung: Claim ≠ Concept (nie beides aus einem Satz)
2) Relations aus Sprach-Triggern (statt pauschalem Fallback)
3) Genau ein META-Frame pro Essay

- Read-first
- Deterministisch (hash-basierte IDs)
- Python stdlib only
"""

import sys
import hashlib
import datetime
import re

UTC = datetime.datetime.utcnow().replace(microsecond=0).isoformat() + "Z"
COMPILER_VERSION = "0.1.1"

# ---------- Helpers ----------

def sha_id(s, prefix):
    h = hashlib.sha256(s.encode("utf-8")).hexdigest()[:24]
    return f"{prefix}_{h}"

def read_input():
    if len(sys.argv) > 1:
        with open(sys.argv[1], "r", encoding="utf-8") as f:
            return f.read()
    return sys.stdin.read()

def normalize(text):
    return "\n".join(line.rstrip() for line in text.strip().splitlines())

def extract_sentences(text):
    parts = re.split(r'(?<=[.!?])\s+', text)
    return [p.strip() for p in parts if p.strip()]

# ---------- Classification ----------

def starts_with_definition(sentence):
    # definitorisch: "X ist Y"
    return bool(re.match(r'^\s*\w[\w\s-]{0,60}\s+ist\s+', sentence.lower()))

def is_claim(sentence):
    # normative/kausale Trigger
    triggers = [
        "führt zu", "impliziert", "bedeutet", "ist schneller",
        "entsteht", "ermöglicht", "verhindert", "bewirkt"
    ]
    s = sentence.lower()
    return any(t in s for t in triggers)

def relation_from_sentence(sentence):
    s = sentence.lower()
    if any(t in s for t in ["weil", "daher", "deshalb"]):
        return "depends_on"
    if any(t in s for t in ["im gegensatz", "aber", "hingegen"]):
        return "contradicts"
    if any(t in s for t in ["beispiel", "etwa", "z. b."]):
        return "example_of"
    if "definiert" in s:
        return "defines"
    if "impliziert" in s:
        return "implies"
    return None

# ---------- Main ----------

def main():
    raw = read_input()
    text = normalize(raw)
    if not text:
        return

    essay_id = sha_id(text[:256], "essay")

    # META (exactly one)
    title = ""
    for line in text.splitlines():
        if line.startswith("#"):
            title = line.lstrip("#").strip()
            break
    if not title:
        title = extract_sentences(text)[0][:120]

    print("# IAL FrameCode v0.1.1")
    print(f"META {sha_id('META|'+essay_id,'meta')} {UTC} essay={essay_id} "
          f"compiler={COMPILER_VERSION} | {title}")
    print()

    sentences = extract_sentences(text)

    claims = []
    concepts = []

    for i, s in enumerate(sentences):
        # Entscheidung: ENTWEDER Concept ODER Claim
        if starts_with_definition(s):
            # Concept
            term = s.split(" ist ", 1)[0].strip().lower().replace(" ", "_")[:32]
            cid = sha_id(f"CNPT|{essay_id}|{term}|{s}", "cnpt")
            concepts.append(cid)
            print(f"CNPT {cid} {UTC} essay={essay_id} term={term} | {s}")
        elif is_claim(s):
            # Claim
            cid = sha_id(f"CLM|{essay_id}|{s}", "clm")
            claims.append((cid, s))
            print(f"CLM {cid} {UTC} essay={essay_id} | {s}")

    # Relations aus Sprache
    for cid, s in claims:
        rel = relation_from_sentence(s)
        if rel and concepts:
            target = concepts[0]  # minimal: erster relevanter Concept-Anker
            rid = sha_id(f"REL|{cid}|{target}|{rel}", "rel")
            print(f"REL {rid} {UTC} essay={essay_id} from={cid} to={target} rel={rel}")

if __name__ == "__main__":
    main()
