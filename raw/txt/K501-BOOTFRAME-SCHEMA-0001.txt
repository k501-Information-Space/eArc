t: SPEC
id: K501-BOOTFRAME-SCHEMA-0001
spec: BOOTFRAME_SCHEMA_V1
kernel: K501
mode: DEVELOPMENT
visibility: LOCAL
canon: READ_ONLY
constraints:
  - APPEND_ONLY
  - NO_PHANTOM
  - NO_IMPLICIT_CANON

goal:
  description: >
    Definiert ein einheitliches Schema für Bootframes inkl. ID- und Versionsregeln,
    sowie Profile-Varianten (LAB, DEVELOPMENT, INCOGNITO).

definitions:
  bootframe:
    meaning: >
      Ein Bootframe ist ein formaler Startzustand/Profilträger für K501.
      Bootframes sind deklarativ, append-only, und beschreiben Gates/Regeln/Scope.
    non_goals:
      - "Kein implizites Ausführen/Deployen/Mutieren"
      - "Kein Truth-Claim ohne Proof"
      - "Keine Canon-Mutation durch Bootframe"

id_and_versioning:
  id_format:
    required: true
    format: "K501-BOOTFRAME-<PROFILE>-V<MAJOR>.<MINOR>-<SEQ4>"
    charset: "ASCII_UPPERCASE_DIGITS_DASH_DOT"
    examples:
      - "K501-BOOTFRAME-DEVELOPMENT-V1.0-0001"
      - "K501-BOOTFRAME-LAB-V1.0-0001"
      - "K501-BOOTFRAME-INCOGNITO-V1.0-0001"
  seq_rules:
    seq4: "0001..9999"
    monotonic: true
    scope: "per PROFILE stream"
  version_rules:
    v_major:
      meaning: "Breaking schema/profile change"
      increment_when:
        - "Neues Pflichtfeld"
        - "Semantik der Gates ändert sich"
        - "ID-Format ändert sich"
    v_minor:
      meaning: "Non-breaking refinement"
      increment_when:
        - "Neue optionale Felder"
        - "Neue erlaubte Gate-States ohne Semantikbruch"
  alias_rules:
    allowed: true
    purpose: "Menschenlesbarer Kurzname"
    examples:
      - "K501 Bootframe 01"
      - "K501 Bootframe LAB"
      - "K501 Bootframe DEVELOPMENT"
      - "Incognito Bootframe"
