<?php
declare(strict_types=1);

header('Content-Type: application/json; charset=utf-8');

$BASE = dirname(__DIR__);
$FRAMES_DIR = $BASE . '/frames';
$LOG_FILE = $FRAMES_DIR . '/frames.ndjson';
$COUNTERS_FILE = $FRAMES_DIR . '/counters.json';

$counts = ['visits' => 0, 'views' => 0, 'clicks' => 0];
$last = null;

if (file_exists($COUNTERS_FILE)) {
  $c = json_decode((string)file_get_contents($COUNTERS_FILE), true);
  if (is_array($c)) {
    $counts['visits'] = (int)($c['visits'] ?? 0);
    $counts['views']  = (int)($c['views'] ?? 0);
    $counts['clicks'] = (int)($c['clicks'] ?? 0);
    $last = $c['last'] ?? null;
  }
}

// last 10 frames (tail)
$lastFrames = [];
if (file_exists($LOG_FILE)) {
  $lines = @file($LOG_FILE, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES);
  if (is_array($lines)) {
    $tail = array_slice($lines, max(0, count($lines) - 10));
    foreach ($tail as $ln) {
      $obj = json_decode($ln, true);
      if (is_array($obj)) {
        $lastFrames[] = [
          'type' => $obj['type'] ?? null,
          't'    => $obj['t'] ?? null,
          'path' => $obj['path'] ?? null,
        ];
      }
    }
    $lastFrames = array_reverse($lastFrames); // newest first
  }
}

echo json_encode([
  'counts' => $counts,
  'last' => $last,
  'lastFrames' => $lastFrames
], JSON_UNESCAPED_SLASHES);
