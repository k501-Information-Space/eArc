# --- CORE ENGINE LOGIC ---
class K501Core:
    def __init__(self, manifest=K501_MANIFEST):
        self.manifest = manifest
        self.pack_hash = manifest["integrity"]

    # [0300] Epistemic Metrics
    def get_mu(self, p):
        p = np.array(p)
        p = p[p > 0]
        return -np.sum(p * np.log2(p))

    def calculate_delta(self, p1, p2):
        p1, p2 = np.array(p1), np.array(p2)
        m = 0.5 * (p1 + p2)
        def kl(p, q): return np.sum(np.where(p > 0, p * np.log2(p / q), 0))
        return 0.5 * kl(p1, m) + 0.5 * kl(p2, m)

    # [0500] Secure Exchange
    def get_seed(self):
        anchor = time.strftime("%Y-%m-%d-%H")
        return int(hashlib.sha256(f"{self.pack_hash}-{anchor}".encode()).hexdigest(), 16) % (2**32)

    def export_ghost(self, p, epsilon=0.01):
        np.random.seed(self.get_seed())
        p_noise = np.abs(np.array(p) + np.random.laplace(0, epsilon, len(p)))
        p_noise /= p_noise.sum()
        indices = np.arange(len(p_noise))
        np.random.shuffle(indices)
        return base64.b64encode(p_noise[indices].tobytes()).decode('utf-8')

    def verify_remote(self, ghost_b64, local_p):
        remote_p = np.frombuffer(base64.b64decode(ghost_b64), dtype=np.float64)
        np.random.seed(self.get_seed())
        idx = np.arange(len(local_p)); np.random.shuffle(idx)
        return self.calculate_delta(remote_p, np.array(local_p)[idx])
