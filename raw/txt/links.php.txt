<?php
// /api/links.php
// K501-GW-ALPHA-1 · Link Parser (Server-Side)
// Reads bootblocks/bootblock.ini and returns JSON

declare(strict_types=1);
header('Content-Type: application/json; charset=utf-8');
header('Cache-Control: public, max-age=300'); // Cache for 5 minutes
header('X-Content-Type-Options: nosniff');

// GET only
if ($_SERVER['REQUEST_METHOD'] !== 'GET') {
    http_response_code(405);
    echo json_encode(['error' => 'METHOD_NOT_ALLOWED']);
    exit;
}

// ============================================
// PARSE INI FILE
// ============================================
$iniPath = __DIR__ . '/../bootblocks/bootblock.ini';

if (!file_exists($iniPath) || !is_readable($iniPath)) {
    http_response_code(404);
    echo json_encode(['error' => 'INI_NOT_FOUND']);
    exit;
}

$raw = parse_ini_file($iniPath, true, INI_SCANNER_TYPED);
if ($raw === false) {
    http_response_code(500);
    echo json_encode(['error' => 'INI_PARSE_ERROR']);
    exit;
}

// ============================================
// FILTER & SANITIZE
// ============================================
$links = [];

foreach ($raw as $key => $link) {
    // Skip hidden links
    if (isset($link['hidden']) && $link['hidden'] === true) {
        continue;
    }

    // Required fields
    if (empty($link['label']) || empty($link['href'])) {
        continue;
    }

    // Sanitize
    $sanitized = [
        'id' => preg_replace('/[^a-z0-9_-]/i', '', $key),
        'label' => htmlspecialchars(trim($link['label']), ENT_QUOTES, 'UTF-8'),
        'href' => filter_var(trim($link['href']), FILTER_SANITIZE_URL),
    ];

    // Optional fields
    if (!empty($link['track'])) {
        $sanitized['track'] = preg_replace('/[^a-z0-9_-]/i', '', $link['track']);
    }

    if (!empty($link['target'])) {
        $sanitized['target'] = in_array($link['target'], ['_blank', '_self', '_parent', '_top']) 
            ? $link['target'] 
            : '_blank';
    } else {
        $sanitized['target'] = '_blank';
    }

    if (!empty($link['rel'])) {
        $sanitized['rel'] = htmlspecialchars(trim($link['rel']), ENT_QUOTES, 'UTF-8');
    } else {
        // Auto-apply security for external links
        if (strpos($sanitized['href'], 'http') === 0 && strpos($sanitized['href'], $_SERVER['HTTP_HOST']) === false) {
            $sanitized['rel'] = 'noopener noreferrer';
        }
    }

    if (!empty($link['class'])) {
        $sanitized['class'] = preg_replace('/[^a-z0-9_ -]/i', '', $link['class']);
    }

    $links[] = $sanitized;
}

// ============================================
// RESPONSE
// ============================================
echo json_encode([
    'status' => 'ok',
    'count' => count($links),
    'links' => $links,
    't' => gmdate('Y-m-d\TH:i:s\Z'),
], JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE);
exit;
