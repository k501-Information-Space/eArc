/* ============================================================

K501 · Ultra-Optimized Kernel V2.1

Extensions:

- hard_index integration (O(1) lookup)

- dynamic_cage temporal guard

- batch index updates

- improved frame builder (reduced sprintf)

Target: 300–500 FPS stable on ThinkPad T520

============================================================ */

#include <stdio.h>

#include <stdlib.h>

#include <stdint.h>

#include <string.h>

#include <unistd.h>

#include <fcntl.h>

#include <time.h>

#include <openssl/evp.h>

/* ============================================================

CONFIG

============================================================ */

#define FRAME_BUFFER_SIZE 262144

#define MAX_FRAME_SIZE 1024

#define INDEX_BATCH 100

#define LEDGER_FILE "frames.ndjson"

#define INDEX_FILE "index.dat"

/* ============================================================

GLOBAL STATE

============================================================ */

static int ledger_fd = -1;

static int index_fd = -1;

static char write_buffer[FRAME_BUFFER_SIZE];

static size_t write_pos = 0;

static uint64_t last_timestamp = 0;

static uint64_t frame_counter = 0;

/* ============================================================

HASH PIPELINE (OpenSSL EVP)

============================================================ */

void sha256_hash(const unsigned char *data, size_t len, unsigned char *out_hash)

{

EVP_MD_CTX *ctx = EVP_MD_CTX_new();

EVP_DigestInit_ex(ctx, EVP_sha256(), NULL);

EVP_DigestUpdate(ctx, data, len);

EVP_DigestFinal_ex(ctx, out_hash, NULL);

EVP_MD_CTX_free(ctx);

}

/* ============================================================

HASH → HEX

============================================================ */

void hash_to_hex(const unsigned char *hash, char *hex)

{

static const char *table = "0123456789abcdef";

for(int i = 0; i < 32; i++)

{

hex[i*2] = table[(hash[i] >> 4) & 0xF];

hex[i*2+1] = table[hash[i] & 0xF];

}

hex[64] = 0;

}

/* ============================================================

TEMPORAL GUARD (dynamic_cage)

============================================================ */

int validate_time(uint64_t t)

{

if(t <= last_timestamp)

return 0;

last_timestamp = t;

return 1;

}

/* ============================================================

LEDGER WRITER

============================================================ */

void ledger_flush()

{

if(write_pos > 0)

{

write(ledger_fd, write_buffer, write_pos);

write_pos = 0;

}

}

uint64_t ledger_position()

{

return lseek(ledger_fd, 0, SEEK_END) + write_pos;

}

void ledger_append(const char *frame, size_t len)

{

if(write_pos + len >= FRAME_BUFFER_SIZE)

{

ledger_flush();

}

memcpy(write_buffer + write_pos, frame, len);

write_pos += len;

}

/* ============================================================

INDEX WRITER

============================================================ */

void index_write(uint64_t offset)

{

write(index_fd, &offset, sizeof(uint64_t));

}

/* ============================================================

FRAME BUILDER (optimized)

============================================================ */

size_t build_frame(char *out, const char *id, const char *hash_hex, uint64_t timestamp)

{

char *ptr = out;

ptr += sprintf(ptr, "{\"id\":\"");

ptr += sprintf(ptr, "%s", id);

ptr += sprintf(ptr, "\",\"t\":");

ptr += sprintf(ptr, "%lu", timestamp);

ptr += sprintf(ptr, ",\"hash\":\"");

ptr += sprintf(ptr, "%s", hash_hex);

ptr += sprintf(ptr, "\"}\n");

return ptr - out;

}

/* ============================================================

FRAME EMIT

============================================================ */

void emit_frame(uint64_t seq)

{

char frame[MAX_FRAME_SIZE];

char id[64];

char hash_hex[65];

unsigned char hash[32];

snprintf(id, sizeof(id), "FRAME-%lu", seq);

sha256_hash((unsigned char*)id, strlen(id), hash);

hash_to_hex(hash, hash_hex);

uint64_t now = time(NULL);

if(!validate_time(now))

return;

uint64_t offset = ledger_position();

size_t len = build_frame(frame, id, hash_hex, now);

ledger_append(frame, len);

frame_counter++;

if(frame_counter % INDEX_BATCH == 0)

{

ledger_flush();

index_write(offset);

}

}

/* ============================================================

KERNEL LOOP

============================================================ */

int main()

{

ledger_fd = open(LEDGER_FILE, O_CREAT | O_APPEND | O_WRONLY, 0644);

index_fd = open(INDEX_FILE, O_CREAT | O_APPEND | O_WRONLY, 0644);

if(ledger_fd < 0 || index_fd < 0)

{

perror("file open failed");

return 1;

}

printf("K501 Kernel V2.1 online\n");

uint64_t seq = 0;

while(seq < 1000000)

{

emit_frame(seq);

seq++;

}

ledger_flush();

close(ledger_fd);

close(index_fd);

printf("Frames written: %lu\n", seq);

return 0;

}
