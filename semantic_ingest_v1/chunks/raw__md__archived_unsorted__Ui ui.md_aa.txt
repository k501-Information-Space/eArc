package com.earc.ui.nav

  

import androidx.compose.runtime.Composable

import androidx.navigation.compose.NavHost

import androidx.navigation.compose.composable

import androidx.navigation.compose.rememberNavController

import com.earc.ui.ui.screens.ConsentScreen

import com.earc.ui.ui.screens.SessionScreen

import com.earc.ui.ui.screens.StartScreen

  

object Routes {

  const val START = "start"

  const val CONSENT = "consent"

  const val SESSION = "session"

}

  

@Composable

fun AppNav() {

  val nav = rememberNavController()

  

  NavHost(navController = nav, startDestination = Routes.START) {

    composable(Routes.START) {

      StartScreen(

        onStart = { nav.navigate(Routes.CONSENT) }

      )

    }

    composable(Routes.CONSENT) {

      ConsentScreen(

        onContinue = { nav.navigate(Routes.SESSION) },

        onCancel = { nav.popBackStack(Routes.START, inclusive = false) }

      )

    }

    composable(Routes.SESSION) {

      SessionScreen(

        onEndSession = { nav.popBackStack(Routes.START, inclusive = false) }

      )

    }

  }

}

    val trimmed = userText.trim()
